        // 2. D3.js Karten-Logik
        document.addEventListener('DOMContentLoaded', () => {
            const mapContainer = document.getElementById('map');
            const loader = document.getElementById('loader');
            
            const width = mapContainer.clientWidth;
            const height = mapContainer.clientHeight || 500;

            const svg = d3.select("#map")
                .append("svg")
                .attr("viewBox", `0 0 ${width} ${height}`)
                .attr("preserveAspectRatio", "xMidYMid meet")
                .style("width", "100%")
                .style("height", "100%");

            // Zuverlässige GeoJSON Quelle aus den Anforderungen
            const geoJsonUrl = 'https://raw.githubusercontent.com/isellsoap/deutschlandGeoJSON/main/2_bundeslaender/3_mittel.geo.json';

            d3.json(geoJsonUrl).then(geojson => {
                loader.style.display = 'none';

                // fitSize berechnet automatisch den Maßstab und die Zentrierung, 
                // sodass Deutschland perfekt in die Box (width x height) passt!
                const projection = d3.geoMercator().fitSize([width, height], geojson);
                const pathGenerator = d3.geoPath().projection(projection);

                svg.selectAll("path")
                    .data(geojson.features)
                    .enter()
                    .append("path")
                    .attr("d", pathGenerator)
                    .attr("class", "state-path")
                    .on("mouseover", function(event, d) {
                        // Dynamisches Auslesen des Namens aus dem GeoJSON (verschiedene Key-Möglichkeiten abfangen)
                        let geoName = d.properties.NAME_1 || d.properties.NAME || d.properties.name || d.properties.GEN || "Unbekannt";
                        if (!this.querySelector('title')) {
                            d3.select(this).append('title').text(geoName);
                        }
                    })
                    .on("click", function(event, d) {
                        // 1. Alle Markierungen zurücksetzen
                        d3.selectAll('.state-path').classed('active', false);
                        
                        // 2. Angeklicktes Bundesland gelb hervorheben (Klasse .active siehe CSS)
                        d3.select(this).classed('active', true);
                        d3.select(this).raise(); // Bringt SVG-Element nach vorne (für saubere Border)

                        // 3. Name extrahieren
                        let geoName = d.properties.NAME_1 || d.properties.NAME || d.properties.name || d.properties.GEN;
                        
                        // 4. Robuster Abgleich des Geo-Namens mit unserer JavaScript-Datenbank
                        // Zuerst exakt prüfen, damit "Sachsen-Anhalt" nicht fälschlicherweise als "Sachsen" erkannt wird.
                        let matchedStateKey = Object.keys(stateData).find(key => key === geoName);
                        
                        // Nur wenn kein exakter Treffer gefunden wurde, suchen wir nach Teilwörtern
                        if (!matchedStateKey) {
                            matchedStateKey = Object.keys(stateData).find(key => 
                                (geoName.includes(key) || key.includes(geoName)) && 
                                !(geoName.includes("Sachsen-Anhalt") && key === "Sachsen") // Ausnahme-Regel
                            );
                        }

                        // 5. DOM aktualisieren
                        document.getElementById('default-msg').style.display = 'none';
                        const infoContent = document.getElementById('info-content');
                        infoContent.classList.remove('visible'); // Reset Animation
                        
                        // Kurzer Timeout für erneutes Triggern der CSS Animation
                        setTimeout(() => {
                            document.getElementById('state-title').innerText = matchedStateKey || geoName;
                            
                            if (matchedStateKey && stateData[matchedStateKey]) {
                                document.getElementById('data-gesetz').innerText = stateData[matchedStateKey].gesetz;
                                document.getElementById('data-vor').innerText = stateData[matchedStateKey].voraussetzungen;
                                document.getElementById('data-fristen').innerText = stateData[matchedStateKey].fristen;
                            } else {
                                // Fallback, falls ein Bundesland in der JS-Datei fehlt
                                document.getElementById('data-gesetz').innerText = "Keine Daten hinterlegt.";
                                document.getElementById('data-vor').innerText = "-";
                                document.getElementById('data-fristen').innerText = "-";
                            }
                            
                            infoContent.classList.add('visible');
                        }, 10);
                    });

            }).catch(error => {
                loader.innerText = "Fehler beim Laden der Karte. Bitte prüfen Sie Ihre Internetverbindung.";
                console.error("GeoJSON Load Error:", error);
            });
            
            // Verhindert das Verzerren der Karte beim Ändern der Fenstergröße
            window.addEventListener('resize', () => {
                const newWidth = mapContainer.clientWidth;
                const newHeight = mapContainer.clientHeight || 500;
                svg.attr("viewBox", `0 0 ${newWidth} ${newHeight}`);
                // Hinweis: d3.geoMercator().fitSize muss theoretisch neu aufgerufen werden, 
                // aber viewBox skaliert SVG nativ bereits sehr gut.
            });
        });
